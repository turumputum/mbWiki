## Описание
Устройство предназначено для реализации поворотного интерфейса в интерактивных цифровых системах. Может быть дополнено кнопкой и подсветкой. Количество фиксированных положений и рабочий диапазон может быть настроен диском трещотки.

![[Устройства/_assets/spinner.pdf|spinner]]

## Аппаратный состав устройства
- (SLOT_0) - [[Аппаратные модули/i2c_hw|Модуль I2C]]
- (SLOT_1) - [[Аппаратные модули/button_smartLed_hw|Модуль кнопки с управляемой подсветкой]]


## Программный состав устройства
- (SLOT_0) - [[Программные модули/encoderAS5600_sw|Энкодер AS5600]]
- (SLOT_1) - [[Программные модули/buttonSmartLed_sw|Кнопка с управляемой подсветкой]]

## Схема подключения
![[Устройства/_assets/spinner.svg]]


## Типовой конфиг
```ini
[SYSTEM] 
deviceName = spinner 

[LAN] 
LAN_enable = 1;0-disable, 1-enable 
;Включаем сетевой адаптер
DHCP = 0;0-disable, 1-enable 
;Динамическая адресация выключена
ipAdress = 192.168.0.33 
netMask = 255.255.255.0 
gateWay = 192.168.0.1 

[MQTT] 
mqttBrokerAdress = 192.168.0.1
;Взаимодействие с сервером организовано счерез mqtt брокера

[SLOT_0] 
mode = encoderAS5600 
options = numOfPos:9, absolute
;модуль настроен на 9 фиксированных положений и работает в абсолютном режиме
cross_link = encoder_0:@->ledRing_1/setPos:@
;значение текущего положения передается в модуль ledRing

[SLOT_1] 
mode = button_ledRing
options = numOfPos:9, increment:40
;модуль настроен на 9 фиксированных положений, сорость приращения анимации настроена на 40 едениц
cross_link = button_1:@->ledRing_1:@
;значение кнопки передается в модуль ledRing
```
